            { patterns: { ja: ["予約フォーム","フォームで予約"], en: ["form","reservation form"] }, key: "reservationForm" },
            { patterns: { ja: ["ありがとう","お疲れ","さようなら","bye"], en: ["thanks","thank","bye"] }, key: "closing" }
        ];

        function getResponse(input) {
            const text = input.trim().toLowerCase();
            const r = responses();
            for (const { patterns, key } of keywords) {
                const ps = patterns[lang] || patterns.ja;
                if (ps.some(p => text.includes(p.toLowerCase()))) {
                    const val = key.split('.').reduce((o,k)=>o?.[k], r);
                    return val || r[key];
                }
            }
            return r.unknown;
        }

        const chatMessages = document.getElementById('chatMessages');
        const userInput = document.getElementById('userInput');
        const sendBtn = document.getElementById('sendBtn');

        function addMessage(text, isUser, isHtml = false) {
            const div = document.createElement('div');
            div.className = `message ${isUser ? 'user' : 'bot'}`;
            div.innerHTML = isHtml
                ? text.replace(/class="link"/g, 'style="color:#d4a574;text-decoration:none"')
                      .replace(/class="tel-link"/g, 'style="color:#d4a574"')
                : text.replace(/\n/g, '<br>');
            chatMessages.appendChild(div);
            chatMessages.scrollTop = chatMessages.scrollHeight;
        }

        function addQuickReplies() {
            const r = responses();
            const wrapper = document.createElement('div');
            wrapper.className = 'message bot';
            wrapper.innerHTML = r.moreQuestion;
            const qrDiv = document.createElement('div');
            qrDiv.className = 'quick-replies';
            r.quickReplies.forEach(text => {
                const btn = document.createElement('button');
                btn.className = 'quick-btn';
                btn.textContent = text;
                btn.onclick = () => { userInput.value = text; sendMessage(); };
                qrDiv.appendChild(btn);
            });
            wrapper.appendChild(qrDiv);
            chatMessages.appendChild(wrapper);
            chatMessages.scrollTop = chatMessages.scrollHeight;
        }

        function sendMessage() {
            const text = userInput.value.trim();
            if (!text) return;
            addMessage(text, true);
            userInput.value = '';
            setTimeout(() => {
                const lastBot = chatMessages.querySelectorAll('.message.bot');
                if (lastBot.length) {
                    const last = lastBot[lastBot.length - 1];
                    if (last.querySelector('.quick-replies')) last.remove();
                }
                addMessage(getResponse(text), false, true);
                addQuickReplies();
            }, 400);
        }

        function setLang(l) {
            lang = l;
            document.querySelectorAll('.lang-btn').forEach(b => b.classList.toggle('active', b.dataset.lang === l));
            userInput.placeholder = responses().placeholder;
            const r = responses();
            chatMessages.innerHTML = '';
            addMessage(r.greeting, false, true);
            addQuickReplies();
        }

        document.getElementById('langJa').onclick = () => setLang('ja');
        document.getElementById('langEn').onclick = () => setLang('en');
        sendBtn.onclick = sendMessage;
        userInput.onkeypress = (e) => { if (e.key === 'Enter') sendMessage(); };

        addMessage(responses().greeting, false, true);
        addQuickReplies();
